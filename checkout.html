Voltar para a loja
`; if (checkoutForm) { checkoutForm.style.display = "none"; } updateTotals(); return; } if (checkoutForm) { checkoutForm.style.display = ""; } checkoutCart.forEach( item => { const price = Number(item.price) || 0; const quantity = Number(item.quantity) || 0; const subtotal = price * quantity; const element = document.createElement( "div" ); element.className = "checkout-product"; element.innerHTML = `
${item.name} ${quantity}x R$ ${formatPrice(price)}
R$ ${formatPrice(subtotal)} `; checkoutItems.appendChild( element ); } ); updateTotals(); } // ============================ // TROCAR FRETE // ============================ shippingOptions.forEach( option => { option.addEventListener( "change", updateTotals ); } ); // ============================ // CEP // ============================ const cepInput = document.getElementById("cep"); const cepStatus = document.getElementById("cepStatus"); const streetInput = document.getElementById("street"); } const data = await response.json(); if (streetInput) { streetInput.value = data.street || ""; } if ( neighborhoodInput ) { neighborhoodInput.value = data.neighborhood || ""; } if (cityInput) { cityInput.value = data.city || ""; } if (stateInput) { stateInput.value = data.state || ""; } if (cepStatus) { cepStatus.textContent = "✓ Endereço encontrado"; } } catch (error) { console.error( "Erro ao consultar CEP:", error ); if (cepStatus) { cepStatus.textContent = "CEP não encontrado"; } } } ); } // ============================ // FINALIZAR PEDIDO // ============================ if (checkoutForm) { checkoutForm.addEventListener( "submit", async event => { event.preventDefault(); checkoutCart = getCart(); if ( checkoutCart.length === 0 ) { alert( "Seu carrinho está vazio." ); return; } const formData = new FormData( checkoutForm ); const shipping = getSelectedShipping(); const subtotal = calculateSubtotal(); const order = { orderNumber: createOrderNumber(), createdAt: new Date().toISOString(), customer: { name: formData.get("name"), email: formData.get("email"), phone: formData.get("phone"), cep: formData.get("cep"), street: formData.get("street"), neighborhood: formData.get( "neighborhood" ), number: formData.get("number"), complement: formData.get( "complement" ), city: formData.get("city"), state: formData.get("state") }, products: checkoutCart, shipping, subtotal, total: subtotal + shipping.price }; if (checkoutSubmit) { checkoutSubmit.disabled = true; checkoutSubmit.textContent = "Processando..."; } try { const response = await fetch( "/api/orders", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify( order ) } ); const data = await response.json() .catch( () => ({}) ); if (!response.ok) { console.error( "Erro da API:", data ); throw new Error( data.error || "Erro ao registrar pedido." ); } // Guardar uma cópia local // para a página de confirmação localStorage.setItem( "lastOrder", JSON.stringify( order ) );
